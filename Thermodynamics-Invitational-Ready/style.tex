\newcommand{\thumbtabs}{%
  \ifnum\value{tabno}>0
  \begin{tikzpicture}[remember picture,overlay]
    \foreach \i in {1,...,7}{%
      \pgfmathsetmacro{\ytop}{0.95+(\i-1)*1.27}%
      \ifodd\value{page}
        \coordinate (A) at ($(current page.north east)+(-0.62in,-\ytop in)$);
        \coordinate (B) at ($(current page.north east)+(-0.27in,-\ytop in-1.20in)$);
        \def\rot{-90}%
      \else
        \coordinate (A) at ($(current page.north west)+(0.27in,-\ytop in)$);
        \coordinate (B) at ($(current page.north west)+(0.62in,-\ytop in-1.20in)$);
        \def\rot{90}%
      \fi
      \ifnum\i=\value{tabno}
        \fill[black] (A) rectangle (B);
        \node[rotate=\rot,text=white,font=\footnotesize\bfseries\sffamily]
          at ($(A)!0.5!(B)$) {\i\ \tabname{\i}};
      \else
        \draw[black!35,line width=0.4pt] (A) rectangle (B);
        \node[rotate=\rot,text=black!45,font=\scriptsize\sffamily]
          at ($(A)!0.5!(B)$) {\i\ \tabname{\i}};
      \fi
    }%
  \end{tikzpicture}\fi}
